echo "=== 事件计数 ==="
grep -c "HANG-DEBUG.*ENTER" /root/pcp_debug_clean.log
grep -c "HANG-DEBUG.*DONE" /root/pcp_debug_clean.log
grep -c "QAG-DEBUG.*before" /root/pcp_debug_clean.log
grep -c "QAG-DEBUG.*after" /root/pcp_debug_clean.log
grep -c "GATHER-DEBUG.*before pcp_allgather" /root/pcp_debug_clean.log
grep -c "GATHER-DEBUG.*after pcp_allgather" /root/pcp_debug_clean.log

echo "=== 每个 worker 的 ENTER/DONE ==="
grep "HANG-DEBUG.*ENTER" /root/pcp_debug_clean.log | sed 's/.*(\(Worker_[^ ]*\).*/\1/' | sort | uniq -c | sort -k2
grep "HANG-DEBUG.*DONE" /root/pcp_debug_clean.log | sed 's/.*(\(Worker_[^ ]*\).*/\1/' | sort | uniq -c | sort -k2

echo "=== GATHER before/after per worker ==="
for w in PCP0_TP0 PCP0_TP1 PCP0_TP2 PCP0_TP3 PCP1_TP0 PCP1_TP1 PCP1_TP2 PCP1_TP3 PCP2_TP0 PCP2_TP1 PCP2_TP2 PCP2_TP3 PCP3_TP0 PCP3_TP1 PCP3_TP2 PCP3_TP3; do
  b=$(grep -c "Worker_${w}.*GATHER-DEBUG.*before pcp" /root/pcp_debug_clean.log)
  a=$(grep -c "Worker_${w}.*GATHER-DEBUG.*after pcp" /root/pcp_debug_clean.log)
  echo "$w: before=$b after=$a diff=$((b-a))"
done

echo "=== QAG before/after per worker ==="
for w in PCP0_TP0 PCP0_TP1 PCP0_TP2 PCP0_TP3 PCP1_TP0 PCP1_TP1 PCP1_TP2 PCP1_TP3 PCP2_TP0 PCP2_TP1 PCP2_TP2 PCP2_TP3 PCP3_TP0 PCP3_TP1 PCP3_TP2 PCP3_TP3; do
  b=$(grep -c "Worker_${w}.*QAG-DEBUG.*before" /root/pcp_debug_clean.log)
  a=$(grep -c "Worker_${w}.*QAG-DEBUG.*after" /root/pcp_debug_clean.log)
  echo "$w: before=$b after=$a diff=$((b-a))"
done

echo "=== 从第一个 DONE 开始的日志（第二轮 forward）==="
first_done=$(grep -n "HANG-DEBUG.*DONE" /root/pcp_debug_clean.log | head -1 | cut -d: -f1)
sed -n "${first_done},\$p" /root/pcp_debug_clean.log | grep -E "HANG-DEBUG|QAG-DEBUG|GATHER-DEBUG" | head -300
