# 1. 看 scheduler 调度了几个请求、各自的 token 数
grep -E "num_scheduled_tokens|num_computed_tokens|total_num_scheduled" /path/to/p_node.log | tail -20

# 2. 看 worker 是否卡在 _model_forward
grep -E "PCP-WORKER.*_model_forward|PCP-WORKER.*START|PCP-WORKER.*DONE" /path/to/p_node.log | tail -30

# 3. 看 CP-DEBUG 日志（如果之前修好了 typo 并加了 logger）
grep "CP-DEBUG" /path/to/p_node.log

# 4. 看 TimeoutError 和 dump 信息
grep -E "TimeoutError|Dumping|num_running_reqs" /path/to/p_node.log | tail -20

# 5. 看有没有 OOM 或其他 NPU 错误
grep -iE "OutOfMemory|HCCL|npu.*error|EZ9999" /path/to/p_node.log | tail -10
