import torch, torch_npu, mmap, os
os.makedirs('/dev/hugepages', exist_ok=True)
fd = os.open('/dev/hugepages/test_dma', os.O_CREAT|os.O_RDWR, 0o600)
os.ftruncate(fd, 2*1024*1024)
mm = mmap.mmap(fd, 2*1024*1024, flags=mmap.MAP_SHARED, prot=mmap.PROT_READ|mmap.PROT_WRITE)
ht = torch.frombuffer(mm, dtype=torch.bfloat16, count=1024)
ht[:] = 42.0
# 普通 .to() 拷贝测试
dt = ht.to('npu:0')
print('DMA copy val:', dt[0].cpu().item())
print('Basic DMA: OK')
mm.close(); os.close(fd); os.unlink('/dev/hugepages/test_dma')
